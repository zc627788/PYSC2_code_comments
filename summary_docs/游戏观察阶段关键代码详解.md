# 游戏观察阶段关键代码详解

## 🎯 概述

游戏观察阶段是LLM-PySC2系统的核心环节，实现了**游戏状态 → 摄像头移动 → 单位选择 → 观察收集 → 文本转换**的完整流程。

## 🔄 完整流程代码分析

### 1. 主循环控制 (`llm_pysc2_agent_main.py`)

```python
def step(self, obs):
    """
    主智能体的核心步骤函数
    
    功能：
    - 接收游戏观察
    - 执行各种初始化步骤
    - 管理子智能体的观察收集和LLM查询
    - 返回游戏动作
    
    Args:
        obs: PySC2游戏观察对象
    """
    super(MainAgent, self).step(obs)
    
    # 1. 初始化和摄像头校准 (必要步骤)
    func_call = main_agent_func0(self, obs)
    if func_call is not None:
        return func_call
    
    # 2. 单位分组，添加到相关智能体团队 (必要步骤)
    func_call = main_agent_func1(self, obs)
    if func_call is not None:
        return func_call
    
    # 3. 自动工人管理 (可选步骤)
    func_call = main_agent_func2(self, obs)
    if func_call is not None:
        return func_call
    
    # 4. 自动工人训练 (可选步骤)
    func_call = main_agent_func3(self, obs)
    if func_call is not None:
        return func_call
    
    # 5. 自动团队集结 (可选步骤)
    func_call = main_agent_func4(self, obs)
    if func_call is not None:
        return func_call
    
    # 6. 子智能体数据更新
    for agent_name in self.AGENT_NAMES:
        agent = self.agents[agent_name]
        agent.num_step = self.steps
        agent.main_loop_step = self.main_loop_step
        agent.world_range = self.world_range
        agent.world_x_offset = self.world_x_offset
        agent.world_y_offset = self.world_y_offset
        agent.size_screen = self.size_screen
        agent.size_minimap = self.size_minimap
        agent.update(obs)  # 更新智能体状态
    
    # 7. LLM决策频率控制
    game_time_s = obs.observation.game_loop / 22.4
    if not self.main_loop_lock and game_time_s - self.game_time_last < 1 / self.config.MAX_LLM_DECISION_FREQUENCY:
        return actions.FUNCTIONS.no_op()
    
    # 8. 主循环：观察收集和LLM查询
    while float(time.time()) - t0 < self.config.MAX_LLM_WAITING_TIME:
        agent_name = self.AGENT_NAMES[self.agent_id]
        agent = self.agents[self.AGENT_NAMES[self.agent_id]]
        
        if not agent.enable:
            self.agent_id = (self.agent_id + 1) % len(self.AGENT_NAMES)
            continue
        
        if not self._all_agent_query_llm_finished():
            # 观察收集阶段
            if agent._is_all_my_teams_ready_to_query():
                # 所有团队观察已收集完毕，开始LLM查询
                agent.thread = threading.Thread(target=thread_act, args=(agent, obs))
                agent.thread.start()
                agent.query_llm_times += 1
            else:
                # 获取未观察的团队和单位标签
                team, tag = agent._get_unobsed_team_and_unit_tag()
                
                # 移动摄像头到单位位置
                func_id, func_call = get_camera_func_smart(self, obs, tag, threshold=self.camera_threshold)
                if func_id == 573:
                    return func_call
                
                # 查找团队头部单位
                unit_f = None
                for unit in obs.observation.feature_units:
                    if unit.tag == tag:
                        unit_f = unit
                
                # 选择单位
                if not unit_f.is_selected:
                    if team['select_type'] == 'select':
                        # 单选模式
                        x, y = min(max(0, unit_f.x), self.size_screen), min(max(0, unit_f.y), self.size_screen)
                        func_id, func_call = (2, actions.FUNCTIONS.select_point('select', (x, y)))
                    elif team['select_type'] == 'select_all_type':
                        # 全选同类型单位
                        x, y = min(max(0, unit_f.x), self.size_screen), min(max(0, unit_f.y), self.size_screen)
                        func_id, func_call = (2, actions.FUNCTIONS.select_point('select_all_type', (x, y)))
                    
                    return func_call
                
                # 收集观察数据
                agent.team_unit_obs_list.append(obs)
                team['obs'].append(obs)
                team['pos'].append([unit_f.x, unit_f.y])
                team['unit_tags'].append(tag)
```

### 2. 摄像头移动函数 (`main_agent_funcs.py`)

```python
def get_camera_func_smart(self, obs, tag, threshold=0.15, team=None):
    """
    智能摄像头移动函数
    
    功能：
    - 检查单位是否在屏幕中央
    - 如果不在，移动摄像头到单位位置
    - 使用历史观察位置优化摄像头移动
    
    Args:
        obs: 游戏观察对象
        tag: 单位标签
        threshold: 摄像头移动阈值
        team: 团队信息
    
    Returns:
        tuple: (func_id, func_call) 函数ID和调用
    """
    unit_r = None
    unit_f = None
    
    # 1. 查找单位在观察中的位置
    for unit in obs.observation.feature_units:
        if unit.tag == tag:
            unit_r = unit
            break
    
    if unit_r is None:
        logger.error(f"[ID {self.log_id}] get_camera_func_smart(): unit {hex(tag)} not found")
        return (0, actions.FUNCTIONS.no_op())
    
    # 2. 检查单位是否在屏幕中央
    if (unit_r.is_on_screen and 
        0.25 * self.size_screen < unit_r.x < 0.75 * self.size_screen and 
        0.25 * self.size_screen < unit_r.y < 0.75 * self.size_screen):
        logger.debug(f"[ID {self.log_id}] get_camera_func_smart(): unit in center")
        return (0, actions.FUNCTIONS.no_op())
    
    # 3. 使用历史观察位置优化摄像头移动
    if team is not None and len(team['obs']) > 0:
        for i in range(len(team['obs'])):
            obs_old = team['obs'][i]
            unit_selected_in_obs = False
            
            # 检查单位是否在历史观察中被选中
            for unit_ in obs_old.observation.feature_units:
                if (unit_.tag == tag and unit_.is_selected and 
                    unit_.alliance == features.PlayerRelative.SELF and 
                    unit_.tag in team['unit_tags']):
                    unit_selected_in_obs = True
                    break
            
            if unit_selected_in_obs:
                x, y = team['pos'][i][0], team['pos'][i][1]
                # 检查摄像头是否已经在正确位置
                if (self.last_two_camera_pos[0][0] == self.last_two_camera_pos[1][0] == x and 
                    self.last_two_camera_pos[0][1] == self.last_two_camera_pos[1][1] == y):
                    logger.debug(f"[ID {self.log_id}] camera in correct position")
                    return (0, actions.FUNCTIONS.no_op())
                else:
                    # 移动摄像头到历史位置
                    self.last_two_camera_pos.append([x, y])
                    return (573, actions.FUNCTIONS.llm_pysc2_move_camera((x, y)))
    
    # 4. 标准模式：移动摄像头到单位位置
    logger.debug(f"[ID {self.log_id}] get_camera_func_smart(): standard mode")
    if (unit_r is not None) and unit_f is None:
        x, y = get_camera_xy(self, unit_r.x, unit_r.y)
        self.last_two_camera_pos.append([x, y])
        return (573, actions.FUNCTIONS.llm_pysc2_move_camera((x, y)))
    
    return (0, actions.FUNCTIONS.no_op())
```

### 3. 单位选择函数 (`main_agent_funcs.py`)

```python
def get_select_func_smart(obs, log_id, team_unit_tags, size_screen):
    """
    智能单位选择函数
    
    功能：
    - 根据单位类型和位置智能选择单位
    - 支持单选和全选模式
    - 优化选择效率
    
    Args:
        obs: 游戏观察对象
        log_id: 日志ID
        team_unit_tags: 团队单位标签列表
        size_screen: 屏幕大小
    
    Returns:
        tuple: (func_id, func_call) 函数ID和调用
    """
    # 1. 获取屏幕上的团队单位
    screen_team_units = []
    for unit in obs.observation.feature_units:
        if unit.tag in team_unit_tags and unit.is_on_screen:
            screen_team_units.append(unit)
    
    if len(screen_team_units) == 0:
        return (0, actions.FUNCTIONS.no_op())
    
    # 2. 检查是否已有选中的单位
    selected_units = [unit for unit in screen_team_units if unit.is_selected]
    if len(selected_units) > 0:
        return (0, actions.FUNCTIONS.no_op())
    
    # 3. 选择第一个未选中的单位
    unit_to_select = screen_team_units[0]
    x, y = min(max(0, unit_to_select.x), size_screen), min(max(0, unit_to_select.y), size_screen)
    
    # 4. 根据单位类型选择选择方式
    if unit_to_select.unit_type in WORKER_TYPE:
        # 工人单位：单选
        return (2, actions.FUNCTIONS.select_point('select', (x, y)))
    else:
        # 战斗单位：全选同类型
        return (2, actions.FUNCTIONS.select_point('select_all_type', (x, y)))
```

### 4. 观察收集和文本转换 (`llm_pysc2_agent.py`)

```python
def query(self, obs) -> None:
    """
    LLM智能体查询函数
    
    功能：
    - 收集所有团队的观察数据
    - 将观察转换为文本描述
    - 调用LLM API获取决策
    - 将文本动作转换为PySC2函数
    
    Args:
        obs: 游戏观察对象
    """
    # 1. 设置等待状态
    while self.is_waiting is False:
        with self.lock:
            self.is_waiting = True
    
    # 2. 获取通信输入
    self.get_text_c_inp()
    
    # 3. 将观察转换为文本描述
    text_o = self.get_text_o(obs)
    
    # 4. 根据观察模式获取文本动作
    if self.config.AGENTS[self.name]['llm']['img_rgb']:
        # RGB图像模式
        base64_image = llm_observation.get_img_obs_rgb(self, obs)
        raw_text_a = self.get_text_a(text_o, base64_image=base64_image)
    elif self.config.AGENTS[self.name]['llm']['img_fea']:
        # 特征图模式
        base64_image = llm_observation.get_img_obs_fea(self, obs)
        raw_text_a = self.get_text_a(text_o, base64_image=base64_image)
    else:
        # 纯文本模式
        raw_text_a = self.get_text_a(text_o)
    
    # 5. 将文本动作转换为PySC2函数
    action_lists, action_list_dict = self.get_func_a(raw_text_a)
    
    # 6. 处理通信输出
    self.get_info_c_out(raw_text_a)
    
    # 7. 设置动作列表
    with self.lock:
        self.is_waiting = False
        self.action_lists = action_lists

def get_text_o(self, obs) -> str:
    """
    将观察转换为文本描述
    
    功能：
    - 使用观察转换器将游戏状态转换为文本
    - 记录观察日志
    - 返回LLM可理解的文本描述
    
    Args:
        obs: 游戏观察对象
    
    Returns:
        str: 文本格式的观察描述
    """
    text_o = ''
    
    # 使用观察转换器进行转换
    text_o = self.translator_o.translate(self)
    
    # 记录观察日志
    if self.name not in self.config.AGENTS_ALWAYS_DISABLE and self.enable:
        with open(self.log_dir_path + f"/{self.name}/o.txt", "a", newline='\n') as f:
            print(json.dumps({self.main_loop_step: text_o}), file=f)
    
    self.last_text_o = text_o
    return text_o
```

### 5. 观察转换器 (`llm_observation.py`)

```python
class CombatGroupTranslatorO(BaseTranslatorO):
    """
    战斗组观察转换器
    
    功能：
    - 将游戏观察转换为战斗组智能体可理解的文本
    - 包含游戏信息、单位信息、任务信息等
    - 支持通信功能
    """
    
    def translate(self, agent) -> str:
        """
        转换观察为文本描述
        
        Args:
            agent: LLM智能体对象
        
        Returns:
            str: 文本格式的观察描述
        """
        obs_list = agent.team_unit_obs_list
        text_obs = ""
        
        if not isinstance(obs_list, list) or len(obs_list) < 1:
            return f"obs_list error, no obs found"
        
        # 1. 获取基础观察信息
        obs = obs_list[0]
        game_info = get_game_info(obs, agent)           # 游戏信息
        units_info = get_teams_info_with_knowledge(agent) # 单位信息
        valid_actions = get_valid_actions_from_obs(obs, agent) # 有效动作
        last_action_info = get_last_action_info(agent)   # 上次动作信息
        task_info = get_task_info(agent)                 # 任务信息
        
        # 2. 组合所有信息
        text_obs = game_info + units_info + valid_actions + last_action_info + task_info
        
        # 3. 添加通信信息（如果启用）
        if agent.config.ENABLE_COMMUNICATION:
            communication_info = get_communication_info(agent)
            text_obs += communication_info
        
        # 4. 添加最终指令
        text_obs += f"\n\nGive each team no more than {agent.config.MAX_NUM_ACTIONS} actions, among which " \
                    f"activity release should usually before move and attack."
        
        if agent.config.ENABLE_COMMUNICATION:
            text_obs += f"\nNow, start generating your analysis, actions and communication:"
        else:
            text_obs += f"\nNow, start generating your analysis and actions:"
        
        return text_obs

def get_game_info(obs, agent) -> str:
    """
    获取游戏基础信息
    
    功能：
    - 提取游戏时间、资源、科技等基础信息
    - 格式化为文本描述
    
    Args:
        obs: 游戏观察对象
        agent: 智能体对象
    
    Returns:
        str: 游戏信息文本
    """
    game_info = f"""
Game Info
  Time: {obs.observation.game_loop / 22.4:.0f}:{int((obs.observation.game_loop / 22.4) % 1 * 60):02d}
  Minerals: {obs.observation.player.minerals}
  Vespene Gas: {obs.observation.player.vespene_gas}
  Supply Used: {obs.observation.player.food_used}
  Supply Available: {obs.observation.player.food_cap}
  Army Count: {obs.observation.player.army_count}
  Worker Count: {obs.observation.player.worker_count}
"""
    return game_info

def get_teams_info_with_knowledge(agent) -> str:
    """
    获取团队单位信息（包含知识库）
    
    功能：
    - 提取每个团队的单位信息
    - 包含位置、血量、能量、技能冷却等
    - 添加单位知识库信息
    
    Args:
        agent: 智能体对象
    
    Returns:
        str: 团队信息文本
    """
    teams_info = ""
    
    for team in agent.teams:
        if len(team['obs']) == 0 or len(team['unit_tags']) == 0:
            continue
        
        team_name = f"Team {team['name']}"
        teams_info += f"\n{team_name} Info:\n"
        
        # 获取团队位置
        if len(team['pos']) > 0:
            team_pos = team['pos'][0]
            teams_info += f"  Team minimap position: [{team_pos[0]:.0f}, {team_pos[1]:.0f}]\n"
        
        # 获取控制的单位信息
        teams_info += "  Controlled Team Units:\n"
        for i, obs in enumerate(team['obs']):
            for unit in obs.observation.feature_units:
                if unit.tag in team['unit_tags'] and unit.is_selected:
                    unit_type_name = units.get_unit_type(unit.unit_type)
                    teams_info += f"    Unit: {unit_type_name}    Tag: {hex(unit.tag)}    "
                    teams_info += f"Pos: ({unit.x:.0f}, {unit.y:.0f})    Health: {unit.health}    "
                    if hasattr(unit, 'energy'):
                        teams_info += f"Energy: {unit.energy}    "
                    if hasattr(unit, 'weapon_cooldown'):
                        teams_info += f"Weapon_cooldown: {unit.weapon_cooldown}"
                    teams_info += "\n"
        
        # 获取附近友军单位
        teams_info += "  Nearby Ally units:\n"
        for obs in team['obs']:
            for unit in obs.observation.feature_units:
                if (unit.alliance == features.PlayerRelative.SELF and 
                    unit.tag not in team['unit_tags'] and 
                    unit.is_on_screen):
                    unit_type_name = units.get_unit_type(unit.unit_type)
                    teams_info += f"    Unit: {unit_type_name}    Tag: {hex(unit.tag)}    "
                    teams_info += f"Pos: ({unit.x:.0f}, {unit.y:.0f})    Health: {unit.health}"
                    if hasattr(unit, 'weapon_cooldown'):
                        teams_info += f"    Weapon_cooldown: {unit.weapon_cooldown}"
                    teams_info += "\n"
        
        # 获取附近敌军单位
        teams_info += "  Nearby Enemy units:\n"
        for obs in team['obs']:
            for unit in obs.observation.feature_units:
                if (unit.alliance == features.PlayerRelative.ENEMY and 
                    unit.is_on_screen):
                    unit_type_name = units.get_unit_type(unit.unit_type)
                    teams_info += f"    Unit: {unit_type_name}    Tag: {hex(unit.tag)}    "
                    teams_info += f"Pos: ({unit.x:.0f}, {unit.y:.0f})    Health: {unit.health}\n"
    
    return teams_info
```

## 🎯 关键执行流程总结

### 1. **游戏状态更新**
- 主循环接收PySC2游戏观察
- 更新智能体状态和游戏信息

### 2. **摄像头移动**
- `get_camera_func_smart()` 检查单位是否在屏幕中央
- 如果不在，使用 `llm_pysc2_move_camera()` 移动摄像头
- 使用历史观察位置优化移动效率

### 3. **单位选择**
- `get_select_func_smart()` 根据单位类型选择选择方式
- 工人单位使用单选，战斗单位使用全选同类型
- 使用 `select_point()` 或 `select_rect()` 执行选择

### 4. **观察收集**
- 将选中的单位添加到智能体团队
- 收集屏幕、小地图、单位信息
- 存储观察数据到 `team_unit_obs_list`

### 5. **文本转换**
- `CombatGroupTranslatorO.translate()` 将观察转换为文本
- 包含游戏信息、单位信息、任务信息、有效动作等
- 支持通信功能和多模态观察

### 6. **LLM查询**
- `get_text_a()` 将文本观察发送给LLM
- 支持纯文本、RGB图像、特征图三种模式
- 异步查询提高效率

这个流程实现了从游戏状态到LLM可理解文本的完整转换，为后续的决策制定奠定了基础。
